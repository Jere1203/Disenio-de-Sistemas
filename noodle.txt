class Grupo{ //Mi clase observada
	Asignacion asignacionActual;
	List<Estudiante> integrantes;
	List<Solicitud> solicitudes;
	Bool estaCerrado;
	String nombre;
	GuitabSDK sdk;
	public Grupo(String nombre, GuitabSDK sdk) {
		this.estaCerrado = false;
		this.nombre = nombre;
		this.sdk = sdk;
	}
	
	void agregarEstudiante(Estudiante estudiante){
		if(!estaCerrado) {
			integrantes.add(estudiante);
			integrantes.forEach(e -> e.notificarInscripcion());		
		} else {
			solicitudes.add(new SolicitudInscripcion(estudiante, this));
		}
	}
	void quitarEstudiante(Estudiante estudiante) {
		integrantes.remove(estudiante);
		integrantes.forEach(e -> e.notificarBaja());
	}
	void cerrar(){
		estaCerrado = true;
		integrantes.forEach(e -> e.notificarCierre());
		sdk.crearRepositorioConAcceso(nombre + "_repositorio", integrantes);
	}
	
	void aceptarInscripcion(Estudiante estudiante) {
		integrantes.add(estudiante);
	}
	
	void aceptarBaja(Estudiante estudiante) {
		integrantes.remove(estudiante);
	}
}
